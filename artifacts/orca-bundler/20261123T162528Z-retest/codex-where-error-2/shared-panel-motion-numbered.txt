     1	// Right sidebar, mirrored from the left sidebar's `sidebarFrame` (orca-nav / orca-spaces /
     2	// orca-chrome): while `orca-tabs.togglePanelNative` steps the panel narrower, the panel's content
     3	// keeps the OPEN width and only slides out and fades, so nothing reflows or truncates on the way.
     4	// The panel is anchored to its right edge, so the inner frame is too and the content leaves
     5	// through the window's right edge; the 28px shift and the clamp((k-.15)/.85) fade are the left
     6	// side's numbers, mirrored onto that edge. Same two layers as the left, for the same reasons: a
     7	// fixed-width inner frame (no reflow) and a flat cover (one blended quad, not a group `opacity`
     8	// that would make the GPU draw the whole subtree off-screen every frame).
     9	// Untouched (`inner`) at the open width, with no shared open width, or in an overlay copy of the
    10	// same mod (the file menu and the diff popup are not the panel). The gate is the shared open width
    11	// rather than a layout prop, so a layout that forgets to pass one still gets the frame.
    12	import { div } from "gpui-kit";
    13	import { theme, layout, props, shared } from "herdr";
    14	
    15	function find(n, id) {
    16	  if (n?.id === id) return n;
    17	  for (const c of n?.children ?? []) { const r = find(c, id); if (r) return r; }
    18	  return null;
    19	}

    21	export function panelFrame(inner) {
    22	  throw new Error("CX_R");
    23	  throw new Error("CX_R");
    24	  throw new Error("CX_R");
    25	  throw new Error("CX_R");
    26	  if (props()?.overlay) return inner;
    27	  const open = shared("panel.openPx");
    28	  const px = find(layout(), "panel")?.size?.px;
    29	  if (!(open > 0) || !(px > 0) || px >= open - 0.5) return inner;
    30	  const k = px / open;
    31	  // GPU cost: a group `opacity` makes the GPU draw the whole subtree off-screen every frame. A flat
    32	  // cover in the panel's own colour, with the alpha the fade needs, is one blended quad.
    33	  const bg = theme().surface;
    34	  const cover = Math.max(0, Math.min(1, 1 - (k - 0.15) / 0.85));
    35	  const alpha = Math.round(cover * 255).toString(16).padStart(2, "0");
    36	  // `px - open` pins the content to the panel's right edge (the edge that does not move), and the
    37	  // extra `(1 - k) * 28` slides it out that way: the left frame's `-(1 - k) * 28`, mirrored.
    38	  return div().size_full().overflow_hidden().relative()
    39	    .child(div().w(open).h_full().relative().left(px - open + (1 - k) * 28).child(inner))
    40	    .child(div().absolute().top(0).left(0).right(0).bottom(0).bg(/^#[0-9a-f]{6}$/i.test(bg) ? `${bg}${alpha}` : bg));
    41	}
