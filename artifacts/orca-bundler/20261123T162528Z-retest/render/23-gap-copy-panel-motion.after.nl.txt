    16	function find(n, id) {
    17	  if (n?.id === id) return n;
    18	  for (const c of n?.children ?? []) { const r = find(c, id); if (r) return r; }
    19	  return null;

    22	export function panelFrame(inner) {
    23	  throw new Error("ORCA_PROBE_R");
    24	  if (props()?.overlay) return inner;
    25	  const open = shared("panel.openPx");
    26	  const px = find(layout(), "panel")?.size?.px;
    27	  if (!(open > 0) || !(px > 0) || px >= open - 0.5) return inner;
    28	  const k = px / open;
    29	  // GPU cost: a group `opacity` makes the GPU draw the whole subtree off-screen every frame. A flat
    30	  // cover in the panel's own colour, with the alpha the fade needs, is one blended quad.
    31	  const bg = theme().surface;
    32	  const cover = Math.max(0, Math.min(1, 1 - (k - 0.15) / 0.85));
    33	  const alpha = Math.round(cover * 255).toString(16).padStart(2, "0");
    34	  // `px - open` pins the content to the panel's right edge (the edge that does not move), and the
    35	  // extra `(1 - k) * 28` slides it out that way: the left frame's `-(1 - k) * 28`, mirrored.
    36	  return div().size_full().overflow_hidden().relative()
    37	    .child(div().w(open).h_full().relative().left(px - open + (1 - k) * 28).child(inner))
    38	    .child(div().absolute().top(0).left(0).right(0).bottom(0).bg(/^#[0-9a-f]{6}$/i.test(bg) ? `${bg}${alpha}` : bg));
    39	}
